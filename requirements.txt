# The shared horizontal packages (agent-platform), pinned by GIT TAG — the form that names a
# version. The repository is PRIVATE, so an image build authenticates with an Azure DevOps PAT
# mounted as a BuildKit secret; nothing at runtime reads it.
#
# LOCAL DEVELOPMENT: these install non-editable, so a change in agent-platform is NOT picked up
# until it is tagged and re-installed. To work across both repositories at once, overlay the
# editable copies afterwards — a developer's choice, deliberately not pinned here:
#   pip install -e ../agent-platform/packages/channel-telegram   # …and the others as needed
#
# `pip`, not `uv pip`: channel-telegram declares conversation-core through a [tool.uv.sources]
# path, and uv resolves that against the tag it was installed from, which conflicts with this
# file's own conversation-core pin. pip does not read that table and installs both from the tags
# named here, which is what CI and the Dockerfile run.
conversation-core @ git+https://dev.azure.com/mdtx/Assistants/_git/agent-platform@conversation-core-v0.2.2#subdirectory=packages/conversation-core
channel-telegram @ git+https://dev.azure.com/mdtx/Assistants/_git/agent-platform@channel-telegram-v0.3.0#subdirectory=packages/channel-telegram

# The framework, with the [db] extra that pulls SQLAlchemy for DatabaseSessionService.
google-adk[db]==2.3.0

# PostgreSQL driver (psycopg 3). The same driver serves the sync business queries and ADK's
# async session engine.
psycopg[binary]==3.2.3

# Required by SQLAlchemy's ASYNC engine, which is what ADK's DatabaseSessionService runs on —
# without it every turn 500s at the session store, before the agent is reached. Pinned here
# because SQLAlchemy's own marker does not cover Apple Silicon: it lists aarch64, ppc64le,
# x86_64, amd64, AMD64 and win32, and macOS on an M-series reports `arm64`.
greenlet==3.5.5

# Web layer: the Telegram webhook mounted by `channel-telegram`, and the two pages `queue_http`
# and `mini_app` mount beside it.
fastapi==0.139.0
uvicorn==0.49.0

# The code a client scans. Pure Python with no dependencies of its own and it writes nothing to
# disk, so it runs under `readOnlyRootFilesystem: true` and the rendered symbol is a value the
# deterministic gate asserts. Pinned EXACTLY: tests/test_qr.py holds a digest of its output, and
# a floating pin would move that digest without a commit here.
segno==1.6.6

# Outbound HTTP client for the Bot API. Resolved transitively by google-genai; pinned here as the
# direct dependency it is.
httpx==0.28.1

# The deterministic no-LLM suite — the real regression gate.
pytest==8.3.4
